#ifdef GL_ES
precision mediump float;
#endif

varying vec4 v_fragmentColor;
varying vec2 v_texCoord;

uniform vec2 resolution;
uniform vec2 offset;
uniform vec2 content;
uniform float angle;

void main(void)
{
	vec2 _xy = v_texCoord.xy * resolution.xy;
	vec2 center = content / 2.0;
	_xy -= offset;

	if(_xy.x < 0.0 || _xy.y < 0.0 || _xy.x > content.x || _xy.y > content.y){
        gl_FragColor = texture2D(CC_Texture0, v_texCoord);
        return;
    }
	vec2 t0 = vec2(0.0, -1.0);
	float _ra;
	if(_xy.x < center.x){
		vec2 t1 = normalize(center - _xy);
		float _angle = dot(t0, t1);
		_ra = acos(_angle) + 3.14159265;
	}else{
		vec2 t1 = normalize(_xy - center);
		float _angle = dot(t0, t1);
		_ra = acos(_angle);
	}
	if(_ra > angle){
		discard;
		//return;
	}
	
	gl_FragColor = texture2D(CC_Texture0, v_texCoord);
}
