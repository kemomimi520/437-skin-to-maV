#ifdef GL_ES
precision lowp float;
#endif

varying vec2 v_texcoord;

uniform sampler2D Texture1;

//区别是这里的蒙版反向, 因为来自fbo的纹理可能是材质，也可能是framebuffer，两者原点不一样
void main(){
    gl_FragColor = texture2D(CC_Texture0, v_texcoord );
    gl_FragColor.a *= texture2D(Texture1, vec2(v_texcoord.x, 1.0 - v_texcoord.y)).r;
}
