#ifdef GL_ES
precision mediump float;
#endif

varying vec4 v_fragmentColor;
varying vec2 v_texCoord;

uniform float unit_h;

void main(void)
{
	float unit = fract(v_texCoord.y / unit_h);
	vec2 coord = vec2(v_texCoord.x, unit);
	gl_FragColor = texture2D(CC_Texture0, coord);
	gl_FragColor.a = v_fragmentColor.a;
}
